cmake_minimum_required(VERSION 3.16)
# PlatformIO/ESP-IDF pass IDF_TARGET; ordinary CMake consumers get the library.
set(PICTOCHAT_NATIVE_DEFAULT ON)
if(DEFINED IDF_TARGET)
    set(PICTOCHAT_NATIVE_DEFAULT OFF)
endif()
option(PICTOCHAT_NATIVE "Build portable library and tests without ESP-IDF" ${PICTOCHAT_NATIVE_DEFAULT})
if(PICTOCHAT_NATIVE)
    project(RetroPicto LANGUAGES C)
    add_subdirectory(lib/pictochat)
    enable_testing()
    find_package(Threads)
    file(GLOB test_sources CONFIGURE_DEPENDS tests/test_*.c)
    foreach(source IN LISTS test_sources)
        get_filename_component(name "${source}" NAME_WE)
        # Only the optional ACK contention test needs POSIX pthreads.
        if(name STREQUAL "test_ack_gate" AND NOT CMAKE_USE_PTHREADS_INIT)
            message(STATUS "Skipping pthread ACK contention test on this toolchain")
            continue()
        endif()
        add_executable(${name} "${source}")
        target_link_libraries(${name} PRIVATE pictochat)
        if(MSVC)
            target_compile_options(${name} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${name} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    endforeach()
    if(TARGET test_ack_gate)
        target_link_libraries(test_ack_gate PRIVATE Threads::Threads)
    endif()
else()
    set(EXTRA_COMPONENT_DIRS firmware/esp32)
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(RetroPicto)
endif()
